cmake_minimum_required(VERSION 3.25)
project(wgpupixel_consumer LANGUAGES CXX)

option(WGPUPIXEL_CONSUMER_TEXT "Require and exercise the optional text component" ON)
if(WGPUPIXEL_CONSUMER_TEXT)
    find_package(wgpupixel CONFIG REQUIRED COMPONENTS core text)
else()
    find_package(wgpupixel CONFIG REQUIRED COMPONENTS core)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE wgpupixel::wgpupixel)
if(WGPUPIXEL_CONSUMER_TEXT)
    target_link_libraries(consumer PRIVATE wgpupixel::text)
    target_compile_definitions(consumer PRIVATE WGPUPIXEL_CONSUMER_TEXT)
endif()
enable_testing()
add_test(NAME installed.consumer COMMAND consumer)
set_tests_properties(installed.consumer PROPERTIES TIMEOUT 120)
